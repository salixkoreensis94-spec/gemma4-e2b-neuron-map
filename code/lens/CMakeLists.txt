cmake_minimum_required(VERSION 3.20)
project(neuron_lens LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(BLAS REQUIRED)
find_package(ZLIB REQUIRED)
set(CBLAS_INCLUDE_DIR "" CACHE PATH "CBLAS include directory")
set(LLAMA_SOURCE "" CACHE PATH "User supplied compatible llama.cpp source")
set(LLAMA_LIBRARY_DIR "" CACHE PATH "User supplied built library directory")
add_executable(lens lens.cpp)
target_include_directories(lens PRIVATE ${CBLAS_INCLUDE_DIR} ${LLAMA_SOURCE}/include ${LLAMA_SOURCE}/ggml/include ${LLAMA_SOURCE}/vendor)
target_link_directories(lens PRIVATE ${LLAMA_LIBRARY_DIR})
target_link_libraries(lens PRIVATE llama ggml ggml-base BLAS::BLAS ZLIB::ZLIB)
