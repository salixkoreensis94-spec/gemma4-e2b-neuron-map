cmake_minimum_required(VERSION 3.20)
project(neuron_map_tools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(BLAS REQUIRED)
find_package(ZLIB REQUIRED)
set(LLAMA_SOURCE "" CACHE PATH "User supplied compatible llama.cpp source")
set(LLAMA_LIBRARY_DIR "" CACHE PATH "User supplied built libraries")
set(CBLAS_INCLUDE_DIR "" CACHE PATH "CBLAS header directory")
add_subdirectory(lens)
foreach(tool IN ITEMS reference export_rope_op export_exp_corrections)
  add_executable(${tool} reference/${tool}.cpp)
endforeach()
add_executable(tokenize tokenize.cpp)
add_executable(tokenize_inputs tokenize_inputs.cpp)
foreach(tool IN ITEMS reference export_rope_op tokenize tokenize_inputs)
  target_include_directories(${tool} PRIVATE ${CBLAS_INCLUDE_DIR} ${LLAMA_SOURCE}/include ${LLAMA_SOURCE}/ggml/include ${LLAMA_SOURCE}/ggml/src ${LLAMA_SOURCE}/ggml/src/ggml-cpu ${LLAMA_SOURCE}/vendor)
  target_link_directories(${tool} PRIVATE ${LLAMA_LIBRARY_DIR})
  target_link_libraries(${tool} PRIVATE llama ggml ggml-base ggml-cpu BLAS::BLAS ZLIB::ZLIB)
endforeach()
target_link_libraries(export_exp_corrections PRIVATE ${CMAKE_DL_LIBS})
